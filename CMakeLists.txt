cmake_minimum_required(VERSION 3.16)
project(qutaq-notify VERSION 0.1 LANGUAGES C CXX)  # C is for the code generated by wayland-scanner

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

# Qt >= 6.5: per-window layer-shell integration (each bubble is a layer surface)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets DBus WaylandClient)
# The wl_output of each QScreen (to find the monitor of the active window)
find_package(Qt6 QUIET COMPONENTS GuiPrivate)

# LayerShellQt 5.x (Plasma 5) turns ALL windows into layer surfaces. Hence 6.x only.
find_package(LayerShellQt 6.0 REQUIRED)

qt_add_executable(qutaq-notify
    src/main.cpp
    src/notification.h      src/notification.cpp
    src/server.h            src/server.cpp
    src/bubble.h            src/bubble.cpp
    src/osd.h               src/osd.cpp
    src/icons.h             src/icons.cpp
    src/history.h           src/history.cpp
    src/activewindow.h      src/activewindow.cpp
    src/settings.h
)

# Generates qwayland-*.h / wayland-*-client-protocol.h from the protocol XML
qt6_generate_wayland_protocol_client_sources(qutaq-notify
    FILES ${CMAKE_CURRENT_SOURCE_DIR}/protocols/wlr-foreign-toplevel-management-unstable-v1.xml
)

target_link_libraries(qutaq-notify PRIVATE
    Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::Widgets Qt6::DBus Qt6::WaylandClient
    LayerShellQt::Interface
)

target_compile_definitions(qutaq-notify PRIVATE
    LSQ_VERSION_MAJOR=${LayerShellQt_VERSION_MAJOR}
    LSQ_VERSION_MINOR=${LayerShellQt_VERSION_MINOR}
)

install(TARGETS qutaq-notify RUNTIME DESTINATION bin)
